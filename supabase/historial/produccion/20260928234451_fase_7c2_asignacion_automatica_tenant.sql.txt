begin;

create or replace function asignar_tenant_automatico()
returns trigger
language plpgsql
set search_path = ''
as $$
declare
  v_tenant uuid;
begin
  v_tenant := public.current_tenant_id();
  if v_tenant is null then
    raise exception 'No se pudo determinar el tenant del usuario autenticado; el alta fue rechazada';
  end if;
  new.tenant_id := v_tenant;
  return new;
end;
$$;

create trigger trg_asignar_tenant_automatico before insert on empresas
  for each row execute function asignar_tenant_automatico();
