  -- ---------------------------------------------------------------------
  -- 6a) REQUISITOS DESDE EL REGISTRO DE PAGO
  --     Revisa el expediente COMPLETO (vacíos y formato) solo cuando el
  --     registro queda activo por: alta activa, reactivación, o cambio de
  --     clasificación de un registro activo. Editar sueldo, periodicidad,
  --     notas, etc. no vuelve a exigir nada. Cerrar nunca se bloquea.
  --     No retroactivo: usa las banderas vigentes de la clasificación en el
  --     momento de la operación; no revisa registros ya existentes.
  --     CONCURRENCIA: actualiza SIN EFECTO la fila del expediente (ver abajo).
  -- ---------------------------------------------------------------------
  create or replace function public.validar_requisitos_registro_pago()
  returns trigger language plpgsql set search_path = '' as $fn$
  declare
    v_revisar boolean;
    c record;
    p record;
    faltan text[] := '{}';
  begin
    if not new.activo then
      return new;
    end if;

    if tg_op = 'INSERT' then
      v_revisar := true;
    else
      v_revisar := (new.clasificacion_id is distinct from old.clasificacion_id)
                or (not old.activo);
    end if;
    if not v_revisar then
      return new;
    end if;

    select nombre, requiere_rfc, requiere_curp, requiere_nss, requiere_cp_fiscal
      into c
      from public.clasificaciones_personal
     where id = new.clasificacion_id and tenant_id = new.tenant_id;
    if not found then
      return new; -- la FK compuesta rechaza la clasificación inexistente o de otro tenant
    end if;
    if not (c.requiere_rfc or c.requiere_curp or c.requiere_nss or c.requiere_cp_fiscal) then
      return new;
    end if;

    -- Coordinación con cambios concurrentes al expediente: se actualiza la
    -- fila de la persona SIN cambiar ningún valor. Un simple bloqueo
    -- (FOR UPDATE / FOR NO KEY UPDATE) no basta: en REPEATABLE READ o
    -- SERIALIZABLE, una transacción concurrente que vacíe RFC/CURP/NSS/CP
    -- con su foto de datos anterior no detecta un bloqueo, pero sí una fila
    -- modificada (recibe 40001). En READ COMMITTED, la transacción que
    -- espera vuelve a evaluar la fila y su trigger ve este registro.
    -- Efectos: nueva versión física de la fila; posible evento UPDATE de
    -- Realtime (si está habilitado en personal). Ningún valor cambia.
    update public.personal
       set nombre_completo = nombre_completo
     where id = new.personal_id and tenant_id = new.tenant_id;

    select rfc, curp, nss, cp_fiscal
      into p
      from public.personal
     where id = new.personal_id and tenant_id = new.tenant_id;
    if not found then
      return new; -- la FK compuesta rechaza la persona inexistente o de otro tenant
    end if;

    if c.requiere_rfc       and not public.es_rfc_persona_valido(nullif(p.rfc, ''))  then faltan := faltan || 'RFC'::text; end if;
    if c.requiere_curp      and not public.es_curp_valida(nullif(p.curp, ''))        then faltan := faltan || 'CURP'::text; end if;
    if c.requiere_nss       and not public.es_nss_valido(nullif(p.nss, ''))          then faltan := faltan || 'NSS'::text; end if;
    if c.requiere_cp_fiscal and not public.es_cp_valido(nullif(p.cp_fiscal, ''))     then faltan := faltan || 'Código postal fiscal'::text; end if;

    if array_length(faltan, 1) > 0 then
      raise exception 'Para un registro de pago "%", el expediente debe tener completos y con formato válido: %.',
        c.nombre, array_to_string(faltan, ', ')
        using errcode = 'P0001', hint = 'requisitos_registro_pago';
    end if;
    return new;
  end; $fn$;

  create trigger trg_validar_requisitos_registro_pago before insert or update on public.personal_registros_pago
    for each row execute function public.validar_requisitos_registro_pago();


  -- ---------------------------------------------------------------------
  -- 6b) REQUISITOS DESDE EL EXPEDIENTE (personal) — trigger NUEVO.
  --     Solo en UPDATE y solo si cambió RFC, CURP, NSS o CP fiscal. Si un
  --     registro ACTIVO de la persona exige un campo que cambió, el valor
  --     nuevo debe estar presente y con formato válido (unión de requisitos
  --     de todos los registros activos). Nunca bloquea editar otros campos.
  --     CONCURRENCIA: un UPDATE con trigger BEFORE ROW bloquea la fila de
  --     personal antes de disparar el trigger; la función (VOLATILE) toma
  --     una instantánea nueva y ve registros confirmados mientras esperaba.
  --     El trigger viejo trg_validar_requisitos_clasificacion (7D-6) NO se
  --     toca: solo actúa si personal.clasificacion_id tiene valor.
  -- ---------------------------------------------------------------------
  create or replace function public.validar_requisitos_expediente()
  returns trigger language plpgsql set search_path = '' as $fn$
  declare
    r record;
    faltan text[] := '{}';
  begin
    if tg_op <> 'UPDATE' then
      return new;
    end if;
    if  new.rfc       is not distinct from old.rfc
    and new.curp      is not distinct from old.curp
    and new.nss       is not distinct from old.nss
    and new.cp_fiscal is not distinct from old.cp_fiscal then
      return new;
    end if;

    select coalesce(bool_or(c.requiere_rfc), false)       as req_rfc,
           coalesce(bool_or(c.requiere_curp), false)      as req_curp,
           coalesce(bool_or(c.requiere_nss), false)       as req_nss,
           coalesce(bool_or(c.requiere_cp_fiscal), false) as req_cp,
           string_agg(distinct c.nombre, ', ')            as clasificaciones
      into r
      from public.personal_registros_pago rp
      join public.clasificaciones_personal c
        on c.id = rp.clasificacion_id and c.tenant_id = rp.tenant_id
     where rp.personal_id = new.id
       and rp.tenant_id  = new.tenant_id
       and rp.activo;

    if r.req_rfc  and new.rfc       is distinct from old.rfc       and not public.es_rfc_persona_valido(nullif(new.rfc, ''))  then faltan := faltan || 'RFC'::text; end if;
    if r.req_curp and new.curp      is distinct from old.curp      and not public.es_curp_valida(nullif(new.curp, ''))        then faltan := faltan || 'CURP'::text; end if;
    if r.req_nss  and new.nss       is distinct from old.nss       and not public.es_nss_valido(nullif(new.nss, ''))          then faltan := faltan || 'NSS'::text; end if;
    if r.req_cp   and new.cp_fiscal is distinct from old.cp_fiscal and not public.es_cp_valido(nullif(new.cp_fiscal, ''))     then faltan := faltan || 'Código postal fiscal'::text; end if;

    if array_length(faltan, 1) > 0 then
      raise exception 'No se puede dejar vacío o inválido: %. Lo exige(n) el/los registro(s) de pago activo(s): %.',
        array_to_string(faltan, ', '), r.clasificaciones
        using errcode = 'P0001', hint = 'requisitos_expediente';
    end if;
    return new;
  end; $fn$;

  create trigger trg_validar_requisitos_expediente before update on public.personal
    for each row execute function public.validar_requisitos_expediente();


  -- ---------------------------------------------------------------------
  -- 7) VISTA DE LECTURA con los días efectivos del periodo.
  --    Implementa la regla "dias_periodo NULL = periodicidades.dias_default"
  --    en un solo lugar (interfaz y Resumen la leerán en 7D-6b).
  --    security_invoker = true: la vista se evalúa con los permisos y el
  --    RLS de quien consulta. SIN esto, una vista en Supabase se evalúa
  --    como su propietario y OMITIRÍA el RLS por tenant.
  -- ---------------------------------------------------------------------
  create view public.v_personal_registros_pago
  with (security_invoker = true) as
  select rp.*,
         coalesce(rp.dias_periodo, pe.dias_default) as dias_periodo_efectivo,
         pe.nombre as periodicidad_nombre,
         c.nombre  as clasificacion_nombre
    from public.personal_registros_pago rp
    join public.periodicidades pe
      on pe.id = rp.periodicidad_id and pe.tenant_id = rp.tenant_id
    join public.clasificaciones_personal c
      on c.id = rp.clasificacion_id and c.tenant_id = rp.tenant_id;

  comment on view public.v_personal_registros_pago is '7D-6a: lectura de registros de pago con dias_periodo_efectivo = coalesce(dias_periodo, periodicidades.dias_default). security_invoker: respeta RLS.';


  -- ---------------------------------------------------------------------
  -- 8) PRIVILEGIOS EXPLÍCITOS (mínimo necesario; no dependen de los
  --    privilegios por defecto del esquema public).
  --    - La app solo usa usuarios autenticados: anon no recibe nada.
  --    - authenticated: SELECT/INSERT/UPDATE en la tabla (sin DELETE ni
  --      TRUNCATE) y solo SELECT en la vista.
  --    - Los privilegios solo abren la puerta; qué filas se ven lo decide
  --      RLS: la tabla por sus políticas y la vista, al ser
  --      security_invoker, por las políticas de sus tres tablas base.
  --    - La vista tiene JOIN, así que no es actualizable automáticamente.
  --    - service_role conserva los privilegios por defecto (uso administrativo).
  -- ---------------------------------------------------------------------
  revoke all on table public.personal_registros_pago from public, anon, authenticated;
  grant select, insert, update on table public.personal_registros_pago to authenticated;

  revoke all on table public.v_personal_registros_pago from public, anon, authenticated;
  grant select on table public.v_personal_registros_pago to authenticated;

  -- ---------------------------------------------------------------------
  -- 9) COMPROBACIÓN FINAL DE PRIVILEGIOS.
  --    a) anon y authenticated: privilegio EFECTIVO con has_table_privilege /
  --       has_any_column_privilege (incluye concesiones directas, las
  --       hechas a PUBLIC y las heredadas de roles de los que sea miembro).
  --    b) PUBLIC: consulta directa al catálogo de privilegios. PUBLIC no es
  --       un rol ordinario; en las listas de privilegios (relacl / attacl)
  --       aparece como grantee = 0 al desglosarlas con aclexplode().
  --       Si relacl es NULL se evalúan los privilegios por defecto
  --       (acldefault), que para tablas y vistas no incluyen a PUBLIC.
  --    Si algo no coincide, se lanza una excepción y TODA la migración se
  --    revierte (es una sola sentencia).
  -- ---------------------------------------------------------------------
  if current_setting('server_version_num')::int >= 170000 then
    v_privs := v_privs || 'MAINTAIN'::text;  -- privilegio existente desde PostgreSQL 17
  end if;
  foreach v_priv in array v_privs loop
    -- anon: ningún privilegio efectivo sobre la tabla ni sobre la vista.
    if has_table_privilege('anon', 'public.personal_registros_pago', v_priv) then v_errores := v_errores || ('anon tiene ' || v_priv || ' en la tabla'); end if;
    if has_table_privilege('anon', 'public.v_personal_registros_pago', v_priv) then v_errores := v_errores || ('anon tiene ' || v_priv || ' en la vista'); end if;

    -- authenticated en la tabla: exactamente SELECT, INSERT, UPDATE.
    if v_priv in ('SELECT','INSERT','UPDATE') then
      if not has_table_privilege('authenticated', 'public.personal_registros_pago', v_priv) then v_errores := v_errores || ('authenticated NO tiene ' || v_priv || ' en la tabla'); end if;
    else
      if has_table_privilege('authenticated', 'public.personal_registros_pago', v_priv) then v_errores := v_errores || ('authenticated tiene ' || v_priv || ' en la tabla'); end if;
    end if;

    -- authenticated en la vista: solo SELECT.
    if v_priv = 'SELECT' then
      if not has_table_privilege('authenticated', 'public.v_personal_registros_pago', v_priv) then v_errores := v_errores || 'authenticated NO tiene SELECT en la vista'::text; end if;
    else
      if has_table_privilege('authenticated', 'public.v_personal_registros_pago', v_priv) then v_errores := v_errores || ('authenticated tiene ' || v_priv || ' en la vista'); end if;
    end if;
  end loop;

  -- Privilegios a nivel columna (no los cubre has_table_privilege).
  foreach v_priv in array array['SELECT','INSERT','UPDATE','REFERENCES'] loop
    if has_any_column_privilege('anon', 'public.personal_registros_pago', v_priv) then v_errores := v_errores || ('anon tiene ' || v_priv || ' en columnas de la tabla'); end if;
    if has_any_column_privilege('anon', 'public.v_personal_registros_pago', v_priv) then v_errores := v_errores || ('anon tiene ' || v_priv || ' en columnas de la vista'); end if;
  end loop;

  -- PUBLIC a nivel tabla/vista (catálogo, grantee = 0).
  if exists (select 1
               from pg_class c
               cross join lateral aclexplode(coalesce(c.relacl, acldefault('r', c.relowner))) a
              where c.oid in ('public.personal_registros_pago'::regclass, 'public.v_personal_registros_pago'::regclass)
                and a.grantee = 0) then
    v_errores := v_errores || 'PUBLIC conserva privilegios sobre la tabla o la vista (pg_class.relacl)'::text;
  end if;

  -- PUBLIC a nivel columna (catálogo, grantee = 0; attacl NULL = sin concesiones de columna).
  if exists (select 1
               from pg_attribute att
               cross join lateral aclexplode(att.attacl) a
              where att.attrelid in ('public.personal_registros_pago'::regclass, 'public.v_personal_registros_pago'::regclass)
                and att.attnum > 0 and not att.attisdropped
                and a.grantee = 0) then
    v_errores := v_errores || 'PUBLIC conserva privilegios de columna (pg_attribute.attacl)'::text;
  end if;

  -- La vista debe haberse creado con security_invoker (respeta RLS).
  if not exists (select 1 from pg_class c
                  where c.oid = 'public.v_personal_registros_pago'::regclass
                    and c.reloptions @> array['security_invoker=true']) then
    v_errores := v_errores || 'la vista no tiene security_invoker=true'::text;
  end if;

  -- RLS habilitado en la tabla nueva.
  if not (select relrowsecurity from pg_class where oid = 'public.personal_registros_pago'::regclass) then
    v_errores := v_errores || 'RLS no está habilitado en personal_registros_pago'::text;
  end if;

  if array_length(v_errores, 1) > 0 then
    raise exception '7D-6a revertida: comprobación final fallida: %', array_to_string(v_errores, '; ');
  end if;

end
$migracion_7d6a$;
